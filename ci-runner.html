.github/workflows/build-dev.yaml • Build & Deploy — Dev 
name: Build & Deploy — Dev

on:
  push:
    branches:
      - main

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Build tag version
        id: tag-version
        run: echo "version=${{ github.run_number }}-${GITHUB_SHA::7}" >> $GITHUB_OUTPUT

      - name: Set up QEMU
        uses: docker/setup-qemu-action@v3

      - name: Set up Docker Buildx
        uses: docker/setup-buildx-action@v3

      - name: Login to ghcr.io
        uses: docker/login-action@v3
        with:
          registry: ghcr.io
          username: ${{ github.actor }}
          password: ${{ secrets.MY_GITHUB_TOKEN }}

      - name: Build and Push Image
        uses: docker/build-push-action@v5
        with:
          context: .
          file: ./deployment/Dockerfile.dev
          push: true
          platforms: linux/amd64,linux/arm64
          tags: ghcr.io/phyoewai-io/pos-app:${{ steps.tag-version.outputs.version }}
